## Installation

```sh
bun add fakeborn
```

## Usage

```ts
import { z } from "zod";
import { fake } from "fakeborn";

const schema = z.object({
  name: z.string(),
  email: z.string().email(),
});

const value = fake(schema);
// → { name: "…", email: "…" } — always satisfies `schema`
```
